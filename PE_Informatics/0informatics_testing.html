def prefix(word):
    ans=''
    for char in word:
        if char not in 'aeiou': ans+=char
        else: return ans
    return ans
def nonprefix(word):
    ans=word
    while ans[0] not in 'aeiou': ans=ans[1:]
    return ans
def nontrivial(word):
    if len(word)<=2:
        return False
    elif word.lower() in ["for", "has", "have", "she", "that", "the", "this", "will", "with"]:
        return False
    elif word[0] in 'aeiou':
        return False
    return True
def spoonerism(phrase):
    nt=[]
    for i in range(len(phrase)):
        if nontrivial(phrase[i]):
            nt.append(i)
    ans=''
    for i in range(len(phrase)):
        if i in nt:
            if i==nt[0]: ans+=prefix(phrase[nt[-1]])+nonprefix(phrase[nt[0]])+' '
            else: ans+=prefix(phrase[nt[nt.index(i)-1]])+nonprefix(phrase[i])+' '
        else:
            ans+=f'{phrase[i]} '
    return ans[:-1]

def get_on(on_stage,actors):
    lst=[]
    for i in range(actors):
        if str(i+1) in on_stage: lst.append('X')
        else: lst.append('-')

def generate(i,actors):
    if actors==1:
        if i==0:
            return ['1']
        return []
    lst=[]
    for num in range(actors-1):
        if (i+1-2**(num))%(2**(num+2))<2**(num+1):
            lst.append(str(num+1))
    if (i+1-2**(actors-1))%(2**actors)<2**(actors-1):
        lst.append(str(actors))
    return lst
def play(actors):
    print(' '.join(['-']*actors))
    prev=[]
    for i in range(2**actors):
        on_stage=generate(i,actors)
        if len(on_stage)>len(prev):
            num=0
            while not (str(num) in on_stage and str(num) not in prev):
                num+=1
            print(f'Enter {num} {' '.join(on_stage)}    {get_on(on_stage,actors)}')

def space(disks):
    if len(disks)<=2:
        return 2*max(disks)
    n=len(disks)
    current_vals=[disks[0]]+[0.0]*(n-2)
    for i in range(1,n-1):
        for j in range(1,i+1):
            current_val=current_vals[i-j]+2*math.sqrt((disks[i-j]*disks[i]))
            current_vals[i]=max(current_vals[i],current_val)

# TieBreaker
def calculate(lst,marks):
    tot=0
    for i,num in enumerate(lst): tot+=num*marks[i]
    return tot
def alter(lst):
    new_lst=lst
    for i in range(len(lst)):
        if new_lst[len(new_lst)-1-i]==0:
            new_lst[len(new_lst)-1-i]+=1
            return new_lst
        new_lst[len(new_lst)-1-i]-=1
    return new_lst
def unique(marks):
    pos_mark=set()
    include=[0]*len(marks)
    while include!=[1]*len(marks):
        num=calculate(include,marks)
        if num in pos_mark and num!=0: return False
        else: pos_mark.add(num)
        include=alter(include)
    return True
def tiebreak(n,s,r):
    solutions=tuple()
    current_solution=[1,2]
    while current_solution[0]<s//n+1:
        if len(current_solution)<n-1:
            if not unique(current_solution):
                if sum(current_solution)>=s:
                    current_solution=current_solution[:-1]
                    current_solution[-1]+=1
                    continue
                current_solution[-1]+=1
            else:
                if sum(current_solution)>=s:
                    current_solution=current_solution[:-1]
                    current_solution[-1]+=1
                    continue
                current_solution.append(current_solution[-1]+1)
        else:
            num=s-sum(current_solution)
            if num<0:
                current_solution=current_solution[:-1]
                current_solution[-1]+=1
                continue
            check_lst=current_solution+[num]
            if unique(check_lst): solutions=solutions+(tuple(sorted(check_lst)),)
            current_solution[-1]+=1
    return set(filter(lambda x:(x[-1]/x[0])<=r,solutions))

def generate(k):
    lst=[2**(k-1)]
    last=list(lst)
    for step in range(k-2,-1,-1):
        new_last=set()
        for num in last:
            lst.append(num+2**step)
            lst.append(num-2**step)
            new_last.add(num+2**step)
            new_last.add(num-2**step)
