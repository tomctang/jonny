	a, b, c, m = 0, 0, 0, 2
	nums=[]
	d=a+b+c
	while d<limits : 
		for n in range(1, m):
			a=m*m-n*n 
			b=2*m*n 
			c=m*m+n*n
			d=a+b+c
			if d>limits: 
				break
			print('Sum of triplet :', d)
			nums.append(d)
			if d==1000:
				print('Special')
		m+=1
	return nums

def sieveEratosthenes(num):
    isPrime = [True] * (num + 1)
    isPrime[0] = False
    isPrime[1] = False
    primes = []
    for i in range(2, (num + 1)):
        if isPrime[i]:
            primes.append(i)
            for j in range(2 * i, (num + 1), i):
                isPrime[j] = False
    return primes

def assign(cand, i, j, val):
    """Assign a definite value to cand[i][j], propagate removal from peers.
       Return False on contradiction, True if assignment done/ok.
    """
    cur = cand[i][j]
    if isinstance(cur, int):
        return cur == val
    if val not in cur:
        return False

        if isinstance(cand[r][c], set) and val in cand[r][c]:

Let \(\mathbb{P}\) be the set of prime numbers. The total sum can be expressed as:

\[\begin{split} \text{Total} = \sum_{\substack{p \in \mathbb{P} \\ 5 \le p \le 10^6}} \left[ \left( \big( q(p) - p \big) \cdot \left(10^{d(p)}\right)^{-1} \pmod{q(p)} \right) \cdot 10^{d(p)} + p \right] \end{split}\]

  1. \(p\): A prime number in the range \(5 \le p \le 10^6\).

  2. \(q(p)\): The next prime number strictly greater than \(p\). Mathematically, \(q(p) = \min \{ q \in \mathbb{P} \mid q > p \}\).

  3. \(d(p)\): The number of decimal digits of \(p\), which can be written as \(d(p) = \lfloor \log_{10} p \rfloor + 1\).

  4. \(\left(10^{d(p)}\right)^{-1} \pmod{q(p)}\): The modular multiplicative inverse of \(10^{d(p)}\) modulo \(q(p)\). This is the unique integer \(x \in \{0, 1, \dots, q(p)-1\}\) such that \(x \cdot 10^{d(p)} \equiv 1 \pmod{q(p)}\). (Note: This inverse always exists because \(q(p) \ge 7\) is prime, meaning \(\gcd(10^{d(p)}, q(p)) = 1\).)

Thus, each term in the sum represents the smallest non-negative integer that ends in the exact digits of \(p\) and is perfectly divisible by the next prime \(q(p)\).
